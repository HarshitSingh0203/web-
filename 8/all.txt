        function calculateFactorial(event) {
            event.preventDefault();

    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>Form Validation</title>

    <style>
        form {
            background-color: aqua;
            font-size: large;
            color: white;
            padding: 20px;
            border-radius: 10px;
            max-width: 350px;
            margin: auto;
        }
    </style>

    <script>
        function validationForm() {

            let name = document.getElementById("a").value;
            let mobile = document.getElementById("b").value;
            let email = document.getElementById("e").value;
            let password = document.getElementById("c").value;

            // Name validation
            let namePattern = /^[A-Za-z\s]+$/;

            if (name === "") {
                alert("Name cannot be blank");
                return false;
            }

            if (!namePattern.test(name)) {
                alert("Name must contain alphabets and whitespace only");
                return false;
            }

            // Mobile number validation
            let mobilePattern = /^[0-9]{10}$/;

            if (!mobilePattern.test(mobile)) {
                alert("Mobile number must contain exactly 10 digits");
                return false;
            }

            // Email validation
            let emailPattern = /^[^@]+@[^@]+$/;

            if (!emailPattern.test(email)) {
                alert("Email must contain exactly one @");
                return false;
            }

            if (!email.endsWith("@united.ac.in")) {
                alert("Email domain must be united.ac.in");
                return false;
            }

            // Password validation
            let passwordPattern =
                /^(?=.*[A-Za-z])(?=.*[0-9])(?=.*[!@#$%&*]).+$/;

            if (!passwordPattern.test(password)) {
                alert("Password must contain at least one alphabet, one digit and one special character (!@#$%&*)");
                return false;
            }

            alert("Form submitted successfully!");
            return true;
        }
    </script>
</head>

<body>

    <form onsubmit="return validationForm();" method="post">

        Name:
        <input type="text" id="a">
        <br><br>

        Mobile Number:
        <input type="text" id="b">
        <br><br>

        Password:
        <input type="password" id="c">
        <br><br>

        Email ID:
        <input type="text" id="e">
        <br><br>

        <input type="submit" value="Submit">

    </form>

</body>

    <script>
        function loadData() {

            let xhttp = new XMLHttpRequest();

            xhttp.onreadystatechange = function() {

                if (this.readyState == 4 && this.status == 200) {
                    document.getElementById("data").innerHTML =
                        this.responseText;
                }
            };

            xhttp.open("GET", "data.txt", true);
            xhttp.send();
        }
    </script>
</head>

<body>

    <h2>AJAX Example</h2>

    <button onclick="loadData()">Load Data</button>

    <div id="data">
        Data will appear here...
    </div>

</body>
</html>
